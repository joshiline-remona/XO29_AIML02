@st.cache_data
def load_data(f_upload):

    if f_upload is not None:

        df = pd.read_excel(
            f_upload
        )

    else:

        df = pd.read_excel(
            "xo.xlsx"
        )

    # Normalize column names
    df.columns = [
        str(c).strip()
        for c in df.columns
    ]

    # Timestamp conversion
    if "timestamp" in df.columns:

        df["timestamp"] = pd.to_datetime(
            df["timestamp"],
            errors="coerce"
        )

        df = df.sort_values(
            "timestamp"
        ).reset_index(
            drop=True
        )

    # Target conversion
    if "target" in df.columns:

        df["target"] = pd.to_numeric(
            df["target"],
            errors="coerce"
        )

    return df

@st.cache_resource(
    show_spinner=
    "Running AERIS End-to-End Intelligence Loop..."
)
def run_aeris_pipeline(
    df_in,
    w_up,
    p_th,
    m_conf,
    f_win,
    c_win,
    c_cov
):

    b_cfg = BOCPDConfig(
        warmup_steps=w_up,
        p_change_threshold=p_th,
        min_confirmation_steps=m_conf
    )

    r_cfg = RegimeConfig()

    pipe = AERISPipeline(
        bocpd_config=b_cfg,
        regime_config=r_cfg,
        forecasting_window=f_win,
        calibration_window=c_win,
        conformal_coverage=c_cov
    )

    return pipe.run_full(
        df_in,
        target_col="target"
    )

        try:

            diffs = (
                raw_df["timestamp"]
                .dropna()
                .sort_values()
                .diff()
                .dropna()
            )

            if len(diffs) > 0:

                median_days = (
                    diffs
                    .dt.total_seconds()
                    .median()
                    / 86400
                )

                if (
                    0.8
                    <= median_days
                    <= 1.2
                ):

                    frequency = "Daily"

                elif (
                    6
                    <= median_days
                    <= 8
                ):

                    frequency = "Weekly"

                elif (
                    27
                    <= median_days
                    <= 32
                ):

                    frequency = "Monthly"

                else:

                    frequency = (
                        f"{median_days:.1f} days"
                    )

            else:

                frequency = "Unknown"

        except Exception:

            frequency = "Unknown"

    else:

        frequency = "Unknown"


    d4.metric(
        "Sampling Frequency",
        frequency
    )
