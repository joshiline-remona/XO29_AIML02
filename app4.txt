@st.cache_data
def load_data(f_upload):

    if f_upload is not None:

        filename = getattr(
            f_upload,
            "name",
            ""
        ).lower()

        if filename.endswith(".csv"):

            df = pd.read_csv(
                f_upload
            )

        else:

            df = pd.read_excel(
                f_upload
            )

    else:

        if os.path.exists(
            DEFAULT_TRAINING_FILE
        ):

            df = pd.read_csv(
                DEFAULT_TRAINING_FILE
            )

        elif os.path.exists(
            FALLBACK_TRAINING_FILE
        ):

            df = pd.read_excel(
                FALLBACK_TRAINING_FILE
            )

        else:

            raise FileNotFoundError(
                "Could not find training_set.csv or xo.xlsx."
            )

    # --------------------------------------------------------
    # Normalize column names
    # --------------------------------------------------------

    df.columns = [
        str(c).strip()
        for c in df.columns
    ]

    # --------------------------------------------------------
    # Timestamp conversion
    # --------------------------------------------------------

    if "timestamp" in df.columns:

        df["timestamp"] = pd.to_datetime(
            df["timestamp"],
            errors="coerce"
        )

        df = (
            df
            .sort_values("timestamp")
            .reset_index(drop=True)
        )

    # --------------------------------------------------------
    # Target conversion
    # --------------------------------------------------------

    if "target" in df.columns:

        df["target"] = pd.to_numeric(
            df["target"],
            errors="coerce"
        )

    # --------------------------------------------------------
    # API feature conversion
    # --------------------------------------------------------

    for col in API_FEATURES:

        if col in df.columns:

            df[col] = pd.to_numeric(
                df[col],
                errors="coerce"
            )

    return df

def safe_status_name(detection):

    status = getattr(
        detection,
        "detector_status",
        None
    )

    if status is None:

        status = getattr(
            detection,
            "status",
            "UNKNOWN"
        )

    if hasattr(
        status,
        "name"
    ):

        return status.name

    return str(status).split(".")[-1]


def safe_regime_name(regime_result):

    regime = getattr(
        regime_result,
        "regime",
        "UNKNOWN"
    )

    if hasattr(
        regime,
        "name"
    ):

        return regime.name

    return str(regime).split(".")[-1]

            regime_result,

# ============================================================
# FEATURE ENGINEERING
# ============================================================

def build_features(df):

    data = df.copy()

    # --------------------------------------------------------
    # Timestamp features
    # --------------------------------------------------------

    if "timestamp" in data.columns:

        ts = pd.to_datetime(
            data["timestamp"],
            errors="coerce"
        )

        data["hour"] = ts.dt.hour

        data["day_of_week"] = ts.dt.dayofweek

        data["day"] = ts.dt.day

        data["day_of_month"] = ts.dt.day

        data["month"] = ts.dt.month

        data["day_of_year"] = ts.dt.dayofyear

        data["is_weekend"] = (
            ts.dt.dayofweek >= 5
        ).astype(float)

    # --------------------------------------------------------
    # Lag features
    # --------------------------------------------------------

    if "target" in data.columns:

        for lag in [
            1,
            2,
            3,
            7,
            14,
            30
        ]:

            data[
                f"lag_{lag}"
            ] = (
                data["target"]
                .shift(lag)
            )

        # ----------------------------------------------------
        # IMPORTANT:
        # shift BEFORE rolling.
        #
        # This prevents current target leakage.
        # ----------------------------------------------------

        shifted_target = (
            data["target"]
            .shift(1)
        )

        for window in [
            7,
            14,
            30
        ]:

            data[
                f"rolling_mean_{window}"
            ] = (
                shifted_target
                .rolling(
                    window=window,
                    min_periods=max(
                        3,
                        window // 2
                    )
                )
                .mean()
            )

            data[
                f"rolling_std_{window}"
            ] = (
                shifted_target
                .rolling(
                    window=window,
                    min_periods=max(
                        3,
                        window // 2
                    )
                )
                .std()
            )

    return data


# ============================================================
# TRAIN BEST MODEL
# ============================================================

def train_best_model(
    history_df,
    validation_days=14
):

    if history_df is None:

        return (
            None,
            None,
            [],
            None
        )

    if len(history_df) < 20:

        return (
            None,
            None,
            [],
            None
        )

    feature_df = build_features(
        history_df
    )

    # --------------------------------------------------------
    # Numeric feature discovery
    # --------------------------------------------------------

    feature_columns = []

    for col in feature_df.columns:

        if col in [
            "timestamp",
            "target"
        ]:

            continue

        if pd.api.types.is_numeric_dtype(
            feature_df[col]
        ):

            feature_columns.append(
                col
            )

    if not feature_columns:

        return (
            None,
            None,
            [],
            None
        )

    clean = feature_df[
        feature_columns + ["target"]
    ].replace(
        [np.inf, -np.inf],
        np.nan
    ).dropna()

    if len(clean) < 20:

        return (
            None,
            None,
            [],
            None
        )

    # --------------------------------------------------------
    # Time-based validation
    # --------------------------------------------------------

    val_size = min(
        validation_days,
        max(
            5,
            len(clean) // 4
        )
    )

    if len(clean) <= val_size:

        return (
            None,
            None,
            [],
            None
        )

    train_part = clean.iloc[
        :-val_size
    ]

    val_part = clean.iloc[
        -val_size:
    ]

    X_train = train_part[
        feature_columns
    ]

    y_train = train_part[
        "target"
    ]

    X_val = val_part[
        feature_columns
    ]

    y_val = val_part[
        "target"
    ]

    candidate_evals = []

    best_name = None
    best_mae = np.inf

    # --------------------------------------------------------
    # Benchmark every model
    # --------------------------------------------------------

    for model_name, model in create_models().items():

        try:

            model.fit(
                X_train,
                y_train
            )

            predictions = model.predict(
                X_val
            )

            mae = mean_absolute_error(
                y_val,
                predictions
            )

            rmse = np.sqrt(
                mean_squared_error(
                    y_val,
                    predictions
                )
            )

            selected = (
                mae < best_mae
            )

            if selected:

                best_mae = mae
                best_name = model_name

            candidate_evals.append(
                {
                    "model_name":
                        model_name,

                    "mae":
                        float(mae),

                    "rmse":
                        float(rmse),

                    "selected":
                        selected
                }
            )

        except Exception as e:

            candidate_evals.append(
                {
                    "model_name":
                        model_name,

                    "mae":
                        np.nan,

                    "rmse":
                        np.nan,

                    "selected":
                        False,

                    "error":
                        str(e)
                }
            )

    if best_name is None:

        return (
            None,
            None,
            candidate_evals,
            None
        )

    # --------------------------------------------------------
    # Refit selected model on all available clean history
    # --------------------------------------------------------

    final_model = create_models()[
        best_name
    ]

    final_model.fit(
        clean[feature_columns],
        clean["target"]
    )

    # --------------------------------------------------------
    # Validation predictions
    # --------------------------------------------------------

    selected_validation_model = create_models()[
        best_name
    ]

    selected_validation_model.fit(
        X_train,
        y_train
    )

    validation_predictions = (
        selected_validation_model.predict(
            X_val
        )
    )

    validation_frame = pd.DataFrame(
        {
            "actual":
                y_val.to_numpy(),

            "prediction":
                validation_predictions
        },
        index=val_part.index
    )

    return (
        final_model,
        best_name,
        candidate_evals,
        (
            feature_columns,
            validation_frame
        )
    )


# ============================================================
# ADAPTIVE MODEL TRAINING
# ============================================================

def train_adaptive_model(
    history_df,
    validation_days=14,
    recent_window=14
):

    if history_df is None:

        return (
            None,
            None,
            [],
            None
        )

    if len(history_df) < 20:

        return (
            None,
            None,
            [],
            None
        )

    # --------------------------------------------------------
    # Recent-regime window
    #
    # We don't throw away the whole history.
    # We deliberately train on a recent section after
    # persistent change confirmation.
    # --------------------------------------------------------

    minimum_window = max(
        40,
        validation_days * 4
    )

    requested_window = max(
        minimum_window,
        recent_window * 4
    )

    requested_window = min(
        requested_window,
        len(history_df)
    )

    adaptive_history = (
        history_df
        .tail(requested_window)
        .copy()
    )

    return train_best_model(
        adaptive_history,
        validation_days=min(
            validation_days,
            max(
                5,
                len(adaptive_history) // 4
            )
        )
    )


# ============================================================
# PREDICT API ROW
# ============================================================

def predict_api_row(
    history_df,
    api_row,
    validation_days,
    strategy="BASELINE",
    recent_window=14
):

    if history_df is None:

        return (
            None,
            None,
            [],
            None
        )

    # --------------------------------------------------------
    # Build history for feature generation
    #
    # The current API target is hidden.
    # --------------------------------------------------------

    current_row = pd.DataFrame(
        [api_row]
    )

    current_row["target"] = np.nan

    working_history = pd.concat(
        [
            history_df.copy(),
            current_row
        ],
        ignore_index=True
    )

    # --------------------------------------------------------
    # Select forecasting strategy
    # --------------------------------------------------------

    if strategy == "ADAPTIVE":

        (
            model,
            model_name,
            candidate_evals,
            model_meta
        ) = train_adaptive_model(
            history_df,
            validation_days,
            recent_window
        )

    else:

        (
            model,
            model_name,
            candidate_evals,
            model_meta
        ) = train_best_model(
            history_df,
            validation_days
        )

    if model is None:

        return (
            None,
            None,
            candidate_evals,
            model_meta
        )

    feature_df = build_features(
        working_history
    )

    feature_columns = model_meta[0]

    latest_features = (
        feature_df
        .iloc[-1]
        .copy()
    )

    X_latest = pd.DataFrame(
        [
            latest_features
        ]
    )

    # --------------------------------------------------------
    # Missing feature handling
    # --------------------------------------------------------

    for col in feature_columns:

        if col not in X_latest.columns:

            X_latest[col] = 0.0

    X_latest = X_latest[
        feature_columns
    ]

    # Fill using historical feature medians
    historical_features = (
        feature_df[
            feature_columns
        ]
        .replace(
            [np.inf, -np.inf],
            np.nan
        )
    )

    for col in feature_columns:

        if pd.isna(
            X_latest.iloc[0][col]
        ):

            median_value = (
                historical_features[col]
                .median()
            )

            if pd.isna(
                median_value
            ):

                median_value = 0.0

            X_latest.loc[
                X_latest.index[0],
                col
            ] = median_value

    prediction = float(
        model.predict(
            X_latest
        )[0]
    )

    return (
        prediction,
        model_name,
        candidate_evals,
        model_meta
    )

# VERIFY CHANGE
#
# THIS IS THE NEW SURPRISE 2 CORE.
# ============================================================

def verify_change(
    historical_values,
    current_value,
    change_probability,
    candidate_start,
    candidate_values,
    baseline_mean,
    baseline_std
):

    values = [
        float(v)
        for v in historical_values
        if pd.notna(v)
    ]

    current_value = float(
        current_value
    )

    # --------------------------------------------------------
    # Need a baseline
    # --------------------------------------------------------

    if len(values) < 10:

        return {
            "state": "NORMAL",
            "candidate_start": candidate_start,
            "candidate_values": candidate_values,
            "baseline_mean": baseline_mean,
            "baseline_std": baseline_std,
            "reason":
                "Insufficient history for change verification.",
            "persistence_count": 0,
            "outside_baseline": False,
            "bocpd_support": False
        }

    # --------------------------------------------------------
    # Create baseline ONLY when a candidate does not exist.
    #
    # Once candidate starts, baseline remains frozen.
    # --------------------------------------------------------

    if (
        candidate_start is None
        and baseline_mean is None
    ):

        baseline_values = values[-20:]

        baseline_mean = float(
            np.mean(
                baseline_values
            )
        )

        baseline_std = float(
            np.std(
                baseline_values
            )
        )

        baseline_std = max(
            baseline_std,
            0.5
        )

    # --------------------------------------------------------
    # Frozen baseline
    # --------------------------------------------------------

    lower_bound = (
        baseline_mean
        - 2.0 * baseline_std
    )

    upper_bound = (
        baseline_mean
        + 2.0 * baseline_std
    )

    outside_baseline = (
        current_value < lower_bound
        or
        current_value > upper_bound
    )

    bocpd_support = (
        change_probability
        >= p_thresh
    )

    # --------------------------------------------------------
    # No candidate currently active
    # --------------------------------------------------------

    if candidate_start is None:

        if (
            outside_baseline
            or
            bocpd_support
        ):

            candidate_values = [
                current_value
            ]

            candidate_start = len(
                values
            )

            reason_parts = []

            if outside_baseline:

                reason_parts.append(
                    "observation is outside "
                    "the frozen baseline range"
                )

            if bocpd_support:

                reason_parts.append(
                    "BOCPD P(Change) crossed "
                    "the detection threshold"
                )

            return {
                "state": "CANDIDATE",
                "candidate_start": candidate_start,
                "candidate_values": candidate_values,
                "baseline_mean": baseline_mean,
                "baseline_std": baseline_std,
                "reason":
                    "Change candidate detected because "
                    + " and ".join(reason_parts)
                    + ". Persistence is being checked.",
                "persistence_count": 1,
                "outside_baseline":
                    outside_baseline,
                "bocpd_support":
                    bocpd_support
            }

        return {
            "state": "NORMAL",
            "candidate_start": None,
            "candidate_values": [],
            "baseline_mean": None,
            "baseline_std": None,
            "reason":
                "Observation remains within the normal "
                "baseline and BOCPD has not confirmed a change.",
            "persistence_count": 0,
            "outside_baseline":
                False,
            "bocpd_support":
                False
        }

    # --------------------------------------------------------
    # Candidate already exists
    # --------------------------------------------------------

    candidate_values = list(
        candidate_values
    )

    candidate_values.append(
        current_value
    )

    persistence_count = len(
        candidate_values
    )

    # --------------------------------------------------------
    # Still abnormal
    # --------------------------------------------------------

    if outside_baseline:

        if (
            persistence_count
            >= ps2_persistence
        ):

            return {
                "state": "PERSISTENT",
                "candidate_start": candidate_start,
                "candidate_values":
                    candidate_values,
                "baseline_mean":
                    baseline_mean,
                "baseline_std":
                    baseline_std,
                "reason":
                    (
                        f"Deviation remained outside "
                        f"the frozen baseline for "
                        f"{persistence_count} consecutive "
                        f"observations. Persistent change confirmed."
                    ),
                "persistence_count":
                    persistence_count,
                "outside_baseline":
                    True,
                "bocpd_support":
                    bocpd_support
            }

        return {
            "state": "VERIFYING",
            "candidate_start": candidate_start,
            "candidate_values":
                candidate_values,
            "baseline_mean":
                baseline_mean,
            "baseline_std":
                baseline_std,
            "reason":
                (
                    f"Deviation is still abnormal, but only "
                    f"{persistence_count}/{ps2_persistence} "
                    f"required observations have been confirmed."
                ),
            "persistence_count":
                persistence_count,
            "outside_baseline":
                True,
            "bocpd_support":
                bocpd_support
        }

    # --------------------------------------------------------
    # It returned to normal BEFORE persistence.
    # --------------------------------------------------------

    return {
        "state": "TEMPORARY DISTURBANCE",
        "candidate_start": None,
        "candidate_values": [],
        "baseline_mean": None,
        "baseline_std": None,
        "reason":
            (
                "The unusual observation returned to the "
                "previous baseline before persistence was "
                "confirmed. Existing forecasting strategy "
                "is retained."
            ),
        "persistence_count":
            persistence_count,
        "outside_baseline":
            False,
        "bocpd_support":
            bocpd_support
    }


# ============================================================
# ADAPTATION DECISION
# ============================================================

def decide_adaptation(
    verification_state
):

    if verification_state == "PERSISTENT":

        return (
            True,
            "Persistent change confirmed. "
            "Adaptive forecasting is activated "
            "for subsequent predictions."
        )

    if verification_state == "TEMPORARY DISTURBANCE":

        return (
            False,
            "Temporary disturbance detected. "
            "The forecasting strategy is held steady "
            "to avoid overreacting to a short-lived event."
        )

    if verification_state in [
        "CANDIDATE",
        "VERIFYING"
    ]:

        return (
            False,
            "Change is still being verified. "
            "No forecasting strategy change is made yet."
        )

    return (
        False,
        "No confirmed persistent change. "
        "Existing forecasting strategy is retained."
    )

# ============================================================
# CONFORMAL CALIBRATION
# ============================================================

def calculate_conformal_radius(
    history_df,
    validation_days,
    calibration_window,
    coverage,
    previous_api_errors=None
):

    residuals = []

    # --------------------------------------------------------
    # First use already revealed PS2 errors.
    #
    # Current actual is NOT included.
    # --------------------------------------------------------

    if previous_api_errors:

        residuals.extend(
            [
                abs(float(e))
                for e in previous_api_errors
                if pd.notna(e)
            ]
        )

    # --------------------------------------------------------
    # Training residuals
    # --------------------------------------------------------

    if len(residuals) < 5:

        (
            model,
            model_name,
            candidate_evals,
            model_meta
        ) = train_best_model(
            history_df,
            validation_days
        )

        if (
            model is not None
            and model_meta is not None
        ):

            feature_columns, validation_frame = (
                model_meta
            )

            training_residuals = (
                np.abs(
                    validation_frame["actual"]
                    -
                    validation_frame["prediction"]
                )
                .tolist()
            )

            residuals.extend(
                training_residuals
            )

    if not residuals:

        target_values = (
            history_df["target"]
            .dropna()
            .astype(float)
        )

        if len(target_values) > 1:

            fallback = float(
                target_values.std()
            )

            if not np.isfinite(
                fallback
            ):

                fallback = 5.0

            return max(
                fallback,
                0.5
            )

        return 5.0

    residuals = np.asarray(
        residuals,
        dtype=float
    )

    residuals = residuals[
        np.isfinite(residuals)
    ]

    if len(residuals) == 0:

        return 5.0

    residuals = residuals[
        -calibration_window:
    ]

    n = len(
        residuals
    )

    # --------------------------------------------------------
    # Finite sample conformal quantile
    # --------------------------------------------------------

    quantile_level = min(
        1.0,
        coverage * (
            1.0 + 1.0 / n
        )
    )

    radius = float(
        np.quantile(
            residuals,
            quantile_level
        )
    )

    return max(
        radius,
        0.5
    )


# ============================================================
# MAKE NEXT ROW FOR OFFLINE FORECAST
# ============================================================

def make_offline_next_row(
    training_df
):

    row = {}

    # --------------------------------------------------------
    # Timestamp
    # --------------------------------------------------------

    if "timestamp" in training_df.columns:

        timestamps = (
            pd.to_datetime(
                training_df["timestamp"],
                errors="coerce"
            )
            .dropna()
            .sort_values()
        )

        if len(timestamps) >= 2:

            delta = (
                timestamps
                .diff()
                .dropna()
                .median()
            )

            if pd.isna(
                delta
            ):

                delta = pd.Timedelta(
                    days=1
                )

            row["timestamp"] = (
                timestamps.iloc[-1]
                + delta
            )

        elif len(timestamps) == 1:

            row["timestamp"] = (
                timestamps.iloc[-1]
                + pd.Timedelta(
                    days=1
                )
            )

    # --------------------------------------------------------
    # Forward-fill other available variables.
    # --------------------------------------------------------

    for col in training_df.columns:

        if col in [
            "target",
            "timestamp"
        ]:

            continue

        row[col] = (
            training_df[col]
            .iloc[-1]
        )

    row["target"] = np.nan

    return row

with t_ps2:

    st.markdown(
        "## PS2 SURPRISE CHALLENGE EVALUATION"
    )

    st.write(
        "This section evaluates AERIS against the live "
        "Surprise Challenge stream without changing the "
        "existing offline AERIS dashboard."
    )

    # ========================================================
    # EXPLANATION
    # ========================================================

    st.info(
        """
        **Surprise 2 decision logic**

        1. BOCPD detects whether the observation looks unusual.
        2. A verification layer checks whether the deviation
           persists.
        3. If it returns to baseline before persistence is
           confirmed, it is classified as a temporary disturbance.
        4. If it remains abnormal for the required number of
           observations, it is classified as a persistent change.
        5. Temporary disturbances keep the existing forecasting
           strategy.
        6. Persistent changes activate adaptive forecasting
           using recent-regime data.
        """
    )

    # ========================================================
    # SESSION CONFIGURATION
    # ========================================================

    st.markdown(
        "### 1. SESSION CONFIGURATION"
    )

    sc1, sc2, sc3 = st.columns(3)

    with sc1:

        team_name = st.text_input(
            "Team Name",
            value=st.session_state.api_team_name,
            disabled=st.session_state.api_started
        )

    with sc2:

        stream_options = [
            s
            for s in [
                "sc1",
                "sc2"
            ]
            if s not in
            st.session_state.api_completed_streams
        ]

        if not stream_options:

            stream_options = [
                "sc1",
                "sc2"
            ]

        selected_stream = st.selectbox(
            "Challenge Stream",
            stream_options,
            index=0,
            disabled=st.session_state.api_started
        )

    with sc3:

        st.metric(
            "Required Persistence",
            f"{ps2_persistence} observations"
        )

    # ========================================================
    # START SESSION
    # ========================================================

    if not st.session_state.api_started:

        if st.button(
            "START PS2 SESSION",
            type="primary",
            use_container_width=True
        ):

            if selected_stream in (
                st.session_state.api_completed_streams
            ):

                st.error(
                    "This stream has already been completed "
                    "in this local session."
                )

            else:

                try:

                    with st.spinner(
                        "Starting PS2 session..."
                    ):

                        session_data = (
                            api_start_session(
                                team_name,
                                selected_stream
                            )
                        )

                    # ------------------------------------------------
                    # Save API session
                    # ------------------------------------------------

                    session_id = (
                        session_data.get(
                            "session_id"
                        )
                    )

                    if not session_id:

                        raise RuntimeError(
                            "API did not return session_id."
                        )

                    st.session_state.api_session_id = (
                        session_id
                    )

                    st.session_state.api_stream = (
                        selected_stream
                    )

                    st.session_state.api_team_name = (
                        team_name
                    )

                    st.session_state.api_total_rows = (
                        session_data.get(
                            "total_rows",
                            0
                        )
                    )

                    st.session_state.api_step = 0

                    st.session_state.api_started = True

                    st.session_state.api_done = False

                    st.session_state.api_current_row = None

                    st.session_state.api_last_result = None

                    # ------------------------------------------------
                    # Clear histories
                    # ------------------------------------------------

                    st.session_state.api_rows = []

                    st.session_state.api_actual_history = []

                    st.session_state.api_prediction_history = []

                    st.session_state.api_error_history = []

                    st.session_state.api_abs_error_history = []

                    st.session_state.api_regime_history = []

                    st.session_state.api_adaptation_history = []

                    st.session_state.api_verification_history = []

                    st.session_state.api_change_probability_history = []

                    st.session_state.api_run_length_history = []

                    st.session_state.api_detector_status_history = []

                    st.session_state.api_interval_lower_history = []

                    st.session_state.api_interval_upper_history = []

                    # ------------------------------------------------
                    # Initialize persistent monitoring
                    #
                    # Training data is replayed ONCE here.
                    # ------------------------------------------------

                    initialize_ps2_monitoring(
                        raw_df
                    )

                    st.success(
                        f"{selected_stream.upper()} session started."
                    )

                    st.rerun()

                except Exception as e:

                    st.error(
                        f"Could not start PS2 session: {e}"
                    )

                    if "403" in str(e):

                        st.warning(
                            "The stream may not be open yet. "
                            "Do not repeatedly create sessions. "
                            "Wait until the challenge stream is available."
                        )

    else:

        # ====================================================
        # ACTIVE SESSION
        # ====================================================

        st.success(
            f"""
            Active session:
            `{st.session_state.api_stream.upper()}`

            Session ID:
            `{st.session_state.api_session_id}`

            Total rows:
            `{st.session_state.api_total_rows}`
            """
        )

        # ====================================================
        # GET NEXT ROW
        # ====================================================

        st.markdown(
            "### 2. GET NEXT OBSERVATION"
        )

        if (
            st.session_state.api_current_row
            is None
            and not st.session_state.api_done
        ):

            if st.button(
                "GET NEXT ROW",
                type="primary",
                use_container_width=True
            ):

                try:

                    with st.spinner(
                        "Requesting next observation..."
                    ):

                        next_data = (
                            api_get_next(
                                st.session_state.api_session_id
                            )
                        )

                    # ------------------------------------------------
                    # Store exactly one current row.
                    # ------------------------------------------------

                    row = {}

                    for key, value in next_data.items():

                        row[key] = value

                    # Some API versions may return a nested row.
                    if "data" in next_data and isinstance(
                        next_data["data"],
                        dict
                    ):

                        row = next_data["data"]

                    st.session_state.api_current_row = (
                        row
                    )

                    st.rerun()

                except Exception as e:

                    st.error(
                        f"Could not get next row: {e}"
                    )

        # ====================================================
        # CURRENT HIDDEN OBSERVATION
        # ====================================================

        if (
            st.session_state.api_current_row
            is not None
        ):

            current_row = (
                st.session_state.api_current_row
            )

            st.markdown(
                "### CURRENT OBSERVATION"
            )

            feature_values = {}

            for feature in API_FEATURES:

                if feature in current_row:

                    feature_values[
                        feature
                    ] = current_row[
                        feature
                    ]

            if feature_values:

                feature_cols = st.columns(
                    min(
                        3,
                        len(feature_values)
                    )
                )

                for idx, (
                    feature,
                    value
                ) in enumerate(
                    feature_values.items()
                ):

                    with feature_cols[
                        idx % len(feature_cols)
                    ]:

                        st.metric(
                            feature,
                            f"{float(value):.4f}"
                            if pd.notna(value)
                            else "N/A"
                        )

            st.warning(
                "The target value is hidden until a prediction "
                "is submitted."
            )

            # ========================================================
            # PREDICT
            # ========================================================

            st.markdown(
                "### 3. FORECAST BEFORE ACTUAL"
            )

            strategy_used = (
                st.session_state.api_forecast_strategy
            )

            # --------------------------------------------------------
            # Build history from training + previously revealed API
            # values.
            # --------------------------------------------------------

            history_df = raw_df.copy()

            if (
                st.session_state.api_rows
                and
                st.session_state.api_actual_history
            ):

                revealed_rows = []

                for idx, row_data in enumerate(
                    st.session_state.api_rows
                ):

                    row_copy = dict(
                        row_data
                    )

                    row_copy["target"] = (
                        st.session_state.api_actual_history[
                            idx
                        ]
                    )

                    revealed_rows.append(
                        row_copy
                    )

                if revealed_rows:

                    revealed_df = pd.DataFrame(
                        revealed_rows
                    )

                    history_df = pd.concat(
                        [
                            history_df,
                            revealed_df
                        ],
                        ignore_index=True
                    )

            if st.button(
                "PREDICT & SUBMIT",
                type="primary",
                use_container_width=True
            ):

                try:

                    # ====================================================
                    # STEP A
                    # Forecast using CURRENT strategy.
                    #
                    # Current actual is NOT known yet.
                    # ====================================================

                    with st.spinner(
                        f"Forecasting using {strategy_used} strategy..."
                    ):

                        (
                            prediction,
                            model_name,
                            candidate_evals,
                            model_meta
                        ) = predict_api_row(
                            history_df,
                            current_row,
                            validation_days=forecast_win,
                            strategy=strategy_used,
                            recent_window=ps2_recent_window
                        )

                    if prediction is None:

                        raise RuntimeError(
                            "Could not generate a forecast."
                        )

                    # ====================================================
                    # STEP B
                    # Calculate conformal interval BEFORE actual reveal.
                    # ====================================================

                    radius = (
                        calculate_conformal_radius(
                            history_df,
                            validation_days=forecast_win,
                            calibration_window=calib_win,
                            coverage=conformal_cov,
                            previous_api_errors=
                                st.session_state.api_abs_error_history
                        )
                    )

                    lower_bound = max(
                        0.0,
                        prediction - radius
                    )

                    upper_bound = (
                        prediction + radius
                    )

                    # ====================================================
                    # STEP C
                    # Submit prediction to API.
                    # This reveals actual.
                    # ====================================================

                    with st.spinner(
                        "Submitting prediction..."
                    ):

                        prediction_response = (
                            api_submit_prediction(
                                st.session_state.api_session_id,
                                prediction
                            )
                        )

                    # ====================================================
                    # STEP D
                    # Extract actual.
                    # ====================================================

                    actual = (
                        prediction_response.get(
                            "actual_target"
                        )
                    )

                    if actual is None:

                        actual = (
                            prediction_response.get(
                                "target"
                            )
                        )

                    if actual is None:

                        actual = (
                            prediction_response.get(
                                "actual"
                            )
                        )

                    if actual is None:

                        raise RuntimeError(
                            "API did not return the revealed actual target."
                        )

                    actual = float(
                        actual
                    )

                    # ====================================================
                    # STEP E
                    # ERROR
                    # ====================================================

                    signed_error = (
                        prediction
                        - actual
                    )

                    absolute_error = abs(
                        signed_error
                    )

                    # ====================================================
                    # STEP F
                    # UPDATE BOCPD EXACTLY ONCE.
                    #
                    # No recreation.
                    # No second replay.
                    # ====================================================

                    detector = (
                        st.session_state.api_detector
                    )

                    if detector is None:

                        initialize_ps2_monitoring(
                            raw_df
                        )

                        detector = (
                            st.session_state.api_detector
                        )

                    detection = detector.update(
                        actual
                    )

                    # ====================================================
                    # STEP G
                    # UPDATE REGIME CLASSIFIER ONCE.
                    # ====================================================

                    classifier = (
                        st.session_state.api_classifier
                    )

                    if classifier is None:

                        classifier = (
                            create_ps2_classifier()
                        )

                        st.session_state.api_classifier = (
                            classifier
                        )

                    regime_result = (
                        classifier.classify(
                            detection
                        )
                    )

                    regime_name = (
                        safe_regime_name(
                            regime_result
                        )
                    )

                    regime_confidence = float(
                        getattr(
                            regime_result,
                            "confidence",
                            0.0
                        )
                    )

                    regime_reason = (
                        safe_regime_reason(
                            regime_result
                        )
                    )

                    # ====================================================
                    # STEP H
                    # CHANGE VERIFICATION
                    #
                    # The historical values passed here do NOT include
                    # current actual yet.
                    #
                    # This means current prediction was not influenced
                    # by current actual.
                    # ====================================================

                    previous_verification_state = (
                        st.session_state.api_verification_state
                    )

                    verification = (
                        verify_change(
                            historical_values=(
                                history_df[
                                    "target"
                                ]
                                .dropna()
                                .astype(float)
                                .tolist()
                            ),

                            current_value=actual,

                            change_probability=
                                get_change_probability(
                                    detection
                                ),

                            candidate_start=
                                st.session_state.api_candidate_start,

                            candidate_values=
                                st.session_state.api_candidate_values,

                            baseline_mean=
                                st.session_state.api_baseline_mean,

                            baseline_std=
                                st.session_state.api_baseline_std
                        )
                    )

                    verification_state = (
                        verification[
                            "state"
                        ]
                    )

                    # ----------------------------------------------------
                    # Save verification state
                    # ----------------------------------------------------

                    st.session_state.api_verification_state = (
                        verification_state
                    )

                    st.session_state.api_candidate_start = (
                        verification[
                            "candidate_start"
                        ]
                    )

                    st.session_state.api_candidate_values = (
                        verification[
                            "candidate_values"
                        ]
                    )

                    st.session_state.api_baseline_mean = (
                        verification[
                            "baseline_mean"
                        ]
                    )

                    st.session_state.api_baseline_std = (
                        verification[
                            "baseline_std"
                        ]
                    )

                    # ====================================================
                    # STEP I
                    # ADAPTATION DECISION
                    # ====================================================

                    (
                        should_adapt,
                        adaptation_reason
                    ) = decide_adaptation(
                        verification_state
                    )

                    entered_persistent = (
                        verification_state
                        == "PERSISTENT"
                        and
                        previous_verification_state
                        != "PERSISTENT"
                    )

                    adaptation_triggered = (
                        should_adapt
                        and
                        entered_persistent
                    )

                    # ----------------------------------------------------
                    # Strategy for NEXT prediction
                    # ----------------------------------------------------

                    if should_adapt:

                        next_strategy = (
                            "ADAPTIVE"
                        )

                    else:

                        next_strategy = (
                            st.session_state.api_forecast_strategy
                        )

                    # ----------------------------------------------------
                    # If temporary:
                    # HOLD current strategy.
                    # ----------------------------------------------------

                    if (
                        verification_state
                        == "TEMPORARY DISTURBANCE"
                    ):

                        next_strategy = (
                            st.session_state.api_forecast_strategy
                        )

                    # ----------------------------------------------------
                    # If candidate/verification:
                    # HOLD current strategy.
                    # ----------------------------------------------------

                    if verification_state in [
                        "CANDIDATE",
                        "VERIFYING"
                    ]:

                        next_strategy = (
                            st.session_state.api_forecast_strategy
                        )

                    st.session_state.api_forecast_strategy = (
                        next_strategy
                    )

                    # ====================================================
                    # STEP J
                    # SAVE HISTORIES
                    # ====================================================

                    st.session_state.api_rows.append(
                        current_row
                    )

                    st.session_state.api_actual_history.append(
                        actual
                    )

                    st.session_state.api_prediction_history.append(
                        prediction
                    )

                    st.session_state.api_error_history.append(
                        signed_error
                    )

                    st.session_state.api_abs_error_history.append(
                        absolute_error
                    )

                    st.session_state.api_regime_history.append(
                        regime_name
                    )

                    st.session_state.api_adaptation_history.append(
                        bool(adaptation_triggered)
                    )

                    st.session_state.api_verification_history.append(
                        verification_state
                    )

                    st.session_state.api_change_probability_history.append(
                        get_change_probability(
                            detection
                        )
                    )

                    st.session_state.api_run_length_history.append(
                        get_map_run_length(
                            detection
                        )
                    )

                    st.session_state.api_detector_status_history.append(
                        get_detector_status(
                            detection
                        )
                    )

                    st.session_state.api_interval_lower_history.append(
                        lower_bound
                    )

                    st.session_state.api_interval_upper_history.append(
                        upper_bound
                    )

                    # ====================================================
                    # STEP K
                    # RESULT LOG
                    # ====================================================

                    result_record = {

                        "step":
                            len(
                                st.session_state
                                .api_actual_history
                            ),

                        "timestamp":
                            current_row.get(
                                "timestamp",
                                ""
                            ),

                        "actual":
                            actual,

                        "prediction":
                            prediction,

                        "error":
                            signed_error,

                        "absolute_error":
                            absolute_error,

                        "model_name":
                            model_name,

                        "forecast_strategy_used":
                            strategy_used,

                        "forecast_strategy_next":
                            next_strategy,

                        "lower_bound":
                            lower_bound,

                        "upper_bound":
                            upper_bound,

                        "interval_radius":
                            radius,

                        "coverage":
                            conformal_cov,

                        "change_probability":
                            get_change_probability(
                                detection
                            ),

                        "map_run_length":
                            get_map_run_length(
                                detection
                            ),

                        "detector_status":
                            get_detector_status(
                                detection
                            ),

                        "verification_state":
                            verification_state,

                        "persistence_count":
                            verification[
                                "persistence_count"
                            ],

                        "baseline_mean":
                            verification[
                                "baseline_mean"
                            ],

                        "baseline_std":
                            verification[
                                "baseline_std"
                            ],

                        "outside_baseline":
                            verification[
                                "outside_baseline"
                            ],

                        "bocpd_support":
                            verification[
                                "bocpd_support"
                            ],

                        "regime":
                            regime_name,

                        "regime_confidence":
                            regime_confidence,

                        "regime_reason":
                            regime_reason,

                        "adaptation_triggered":
                            adaptation_triggered,

                        "adaptation_reason":
                            adaptation_reason
                    }

                    st.session_state.api_last_result = (
                        result_record
                    )

                    # ====================================================
                    # STEP L
                    # CHECK COMPLETION
                    # ====================================================

                    rows_processed = len(
                        st.session_state.api_actual_history
                    )

                    total_rows = (
                        st.session_state.api_total_rows
                    )

                    if (
                        total_rows
                        and
                        rows_processed
                        >= total_rows
                    ):

                        st.session_state.api_done = True

                        if (
                            st.session_state.api_stream
                            not in
                            st.session_state.api_completed_streams
                        ):

                            st.session_state.api_completed_streams.append(
                                st.session_state.api_stream
                            )

                    # ====================================================
                    # CLEAR CURRENT ROW
                    # ====================================================

                    st.session_state.api_current_row = None

                    st.rerun()

                except Exception as e:

                    st.error(
                        f"Prediction/evaluation error: {e}"
                    )

        # ====================================================
        # CURRENT PS2 STATE
        # ====================================================

        st.markdown(
            "---"
        )

        st.markdown(
            "### 4. SURPRISE 2 DECISION STATE"
        )

        state = (
            st.session_state.api_verification_state
        )

        if state == "NORMAL":

            st.success(
                """
                NORMAL

                No confirmed deviation is currently active.
                """
            )

        elif state == "CANDIDATE":

            st.warning(
                """
                CANDIDATE CHANGE

                BOCPD/statistical evidence suggests something
                unusual may have happened. Persistence is not
                confirmed yet.
                """
            )

        elif state == "VERIFYING":

            st.warning(
                """
                VERIFYING

                The deviation is still present, but the required
                persistence threshold has not yet been reached.

                Forecasting strategy remains unchanged.
                """
            )

        elif state == "TEMPORARY DISTURBANCE":

            st.info(
                """
                TEMPORARY DISTURBANCE

                The unusual observation returned to the previous
                baseline before persistence was confirmed.

                The existing forecasting strategy was retained.
                """
            )

        elif state == "PERSISTENT":

            st.error(
                """
                PERSISTENT CHANGE

                The deviation remained outside the frozen baseline
                for the required number of observations.

                Adaptive forecasting is now active.
                """
            )

        if st.session_state.api_last_result:

            latest_ps2 = (
                st.session_state.api_last_result
            )

            v1, v2, v3, v4 = st.columns(4)

            v1.metric(
                "P(Change)",
                f"{latest_ps2['change_probability']:.3f}"
            )

            v2.metric(
                "Persistence",
                (
                    f"{latest_ps2['persistence_count']}"
                    f"/{ps2_persistence}"
                )
            )

            v3.metric(
                "Verification",
                latest_ps2[
                    "verification_state"
                ]
            )

            v4.metric(
                "Next Forecast Strategy",
                latest_ps2[
                    "forecast_strategy_next"
                ]
            )

            st.write(
                "**Verification Reason:**"
            )

            st.write(
                latest_ps2[
                    "adaptation_reason"
                ]
            )

            if (
                latest_ps2[
                    "baseline_mean"
                ]
                is not None

    # --------------------------------------------------------

    if "timestamp" in raw_df.columns:

        try:

            diffs = (
                raw_df["timestamp"]
                .dropna()
                .sort_values()
                .diff()
                .dropna()
            )

            if len(diffs) > 0:

                median_days = (
                    diffs
                    .dt.total_seconds()
                    .median()
                    / 86400
                )

                if (
                    0.8
                    <= median_days
                    <= 1.2
                ):

                    frequency = "Daily"

                elif (
                    6
                    <= median_days
                    <= 8
                ):

                    frequency = "Weekly"

                elif (
                    27
                    <= median_days
                    <= 32
                ):

                    frequency = "Monthly"

                else:

                    frequency = (
                        f"{median_days:.1f} days"
                    )

            else:

                frequency = "Unknown"

        except Exception:

            frequency = "Unknown"

    else:

        frequency = "Unknown"

    d4.metric(
        "Sampling Frequency",
        frequency
    )
