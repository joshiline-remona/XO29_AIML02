    )

        data[TARGET].shift(7)
    )

    )

        f"    Usable rows : {len(data)}"
    )

# 3. BAYESIAN-STYLE QUIET SHIFT DETECTOR
# ============================================================

def calculate_change_probability(values):

    values = np.asarray(
        values,
        dtype=float
    )

    required = (
        BASELINE_WINDOW
        +
        DETECTION_WINDOW
    )

    if len(values) < required:

        return {
            "probability": 0.0,
            "mean_shift": 0.0,
            "standardized_shift": 0.0,
            "volatility_ratio": 1.0,
            "recent_mean": 0.0,
            "baseline_mean": 0.0,
            "recent_std": 0.0,
            "baseline_std": 1.0
        }

    # --------------------------------------------------------
    # Historical baseline
    # --------------------------------------------------------

    baseline = values[
        -(BASELINE_WINDOW + DETECTION_WINDOW):
        -DETECTION_WINDOW
    ]

    # --------------------------------------------------------
    # Recent detection window
    # --------------------------------------------------------

    recent = values[
        -DETECTION_WINDOW:
    ]

    baseline_mean = np.mean(
        baseline
    )

    recent_mean = np.mean(
        recent
    )

    baseline_std = np.std(
        baseline
    )

    recent_std = np.std(
        recent
    )

    if baseline_std < 1e-6:
        baseline_std = 1e-6

    # --------------------------------------------------------
    # Signal 1:
    # Mean shift
    # --------------------------------------------------------

    mean_shift = abs(
        recent_mean
        -
        baseline_mean
    )

    standardized_shift = (
        mean_shift
        /
        baseline_std
    )

    # --------------------------------------------------------
    # Signal 2:
    # Volatility shift
    # --------------------------------------------------------

    volatility_ratio = (
        recent_std
        /
        baseline_std
    )

    volatility_evidence = abs(
        np.log(
            max(
                volatility_ratio,
                1e-6
            )
        )
    )

    # --------------------------------------------------------
    # Combined evidence
    # --------------------------------------------------------

    evidence = (
        0.75 * standardized_shift
        +
        0.25 * volatility_evidence
    )

    # Convert evidence to bounded
    # probability-like score.
    probability = (
        1
        -
        np.exp(
            -evidence / 2
        )
    )

    probability = float(
        np.clip(
            probability,
            0,
            0.99
        )
    )

    return {

        "probability":
            probability,

        "mean_shift":
            mean_shift,

        "standardized_shift":
            standardized_shift,

        "volatility_ratio":
            volatility_ratio,

        "recent_mean":
            recent_mean,

        "baseline_mean":
            baseline_mean,

        "recent_std":
            recent_std,

        "baseline_std":
            baseline_std
    }

# 4. STREAMING QUIET SHIFT DETECTION
# ============================================================

def run_streaming_change_detection(data):

    print("\n[3] BAYESIAN QUIET-SHIFT DETECTION")

    print(
        f"    Baseline window  : "
        f"{BASELINE_WINDOW} observations"
    )

    print(
        f"    Detection window : "
        f"{DETECTION_WINDOW} observations"
    )

    print(
        f"    Watch threshold  : "
        f"{WATCH_THRESHOLD}"
    )

    print(
        f"    Change threshold : "
        f"{CHANGE_THRESHOLD}"
    )

    print(
        f"    Persistence      : "
        f"{PERSISTENCE_REQUIRED} observations"
    )

    values = (
        data[TARGET]
        .values
    )

    timestamps = (
        data["timestamp"]
        .values
    )

    persistence_count = 0

    first_genuine_change = None

    detection_history = []

    start_index = (
        BASELINE_WINDOW
        +
        DETECTION_WINDOW
    )

    previous_state = None

    # --------------------------------------------------------
    # Sequential replay
    # --------------------------------------------------------

    for i in range(
        start_index,
        len(values)
    ):

        current_values = (
            values[:i + 1]
        )

        evidence = (
            calculate_change_probability(
                current_values
            )
        )

        probability = (
            evidence["probability"]
        )

        # ----------------------------------------------------
        # Decision
        # ----------------------------------------------------

        if probability >= CHANGE_THRESHOLD:

            persistence_count += 1

            if (
                persistence_count
                >= PERSISTENCE_REQUIRED
            ):

                state = "GENUINE CHANGE"

                if (
                    first_genuine_change
                    is None
                ):

                    first_genuine_change = {

                        "index": i,

                        "timestamp":
                            timestamps[i],

                        **evidence,

                        "persistence":
                            persistence_count
                    }

            else:

                state = "WATCH"

        elif (
            probability
            >= WATCH_THRESHOLD
        ):

            persistence_count = 0

            state = "WATCH"

        else:

            persistence_count = 0

            state = "NORMAL"

        # ----------------------------------------------------
        # Save history
        # ----------------------------------------------------

        detection_history.append({

            "index": i,

            "timestamp":
                timestamps[i],

            "probability":
                probability,

            "state":
                state,

            "persistence":
                persistence_count,

            "mean_shift":
                evidence["mean_shift"],

            "volatility_ratio":
                evidence["volatility_ratio"]
        })

        # ----------------------------------------------------
        # Terminal explanation
        # ----------------------------------------------------

        if state != previous_state:

            print(
                f"\n    {pd.Timestamp(timestamps[i]).date()} "
                f"→ {state}"
            )

            print(
                f"       P(change)       = "
                f"{probability:.3f}"
            )

            print(
                f"       Mean shift      = "
                f"{evidence['mean_shift']:.3f}"
            )

            print(
                f"       Volatility ratio = "
                f"{evidence['volatility_ratio']:.3f}"
            )

            print(
                f"       Persistence     = "
                f"{persistence_count}/"
                f"{PERSISTENCE_REQUIRED}"
            )

        previous_state = state

    # --------------------------------------------------------
    # Final detection report
    # --------------------------------------------------------

    print(
        "\n    ---------------------------------------"
    )

    if first_genuine_change is not None:

        print(
            "    FIRST GENUINE CHANGE DETECTED"
        )

        print(
            f"    Timestamp        : "
            f"{pd.Timestamp(first_genuine_change['timestamp']).date()}"
        )

        print(
            f"    Probability      : "
            f"{first_genuine_change['probability']:.3f}"
        )

        print(
            f"    Mean shift       : "
            f"{first_genuine_change['mean_shift']:.3f}"
        )

        print(
            f"    Volatility ratio : "
            f"{first_genuine_change['volatility_ratio']:.3f}"
        )

        print(
            "\n    WHY WAS THIS NOT TREATED AS NOISE?"
        )

        print(
            "    AERIS waited for persistent evidence"
        )

        print(
            "    instead of reacting to a single point."
        )

    else:

        print(
            "    No genuine persistent change confirmed."
        )

        print(
            "    Isolated fluctuations were ignored."
        )

    return {

        "history":
            detection_history,

        "first_change":
            first_genuine_change
    }

# 5. REGIME DETECTION
# ============================================================

def detect_regime(
    values,
    change_probability
):

    print("\n[4] REGIME DETECTION")

    values = np.asarray(values)

    recent = (
        values[-DETECTION_WINDOW:]
    )

    baseline = (
        values[-BASELINE_WINDOW:]
    )

    recent_mean = np.mean(
        recent
    )

    baseline_mean = np.mean(
        baseline
    )

    recent_std = np.std(
        recent
    )

    baseline_std = np.std(
        baseline
    )

    if baseline_std < 1e-6:

        baseline_std = 1e-6

    mean_shift = (
        recent_mean
        -
        baseline_mean
    )

    volatility_ratio = (
        recent_std
        /
        baseline_std
    )

    print(
        f"    Recent mean      : "
        f"{recent_mean:.3f}"
    )

    print(
        f"    Baseline mean    : "
        f"{baseline_mean:.3f}"
    )

    print(
        f"    Mean shift       : "
        f"{mean_shift:.3f}"
    )

    print(
        f"    Volatility ratio : "
        f"{volatility_ratio:.3f}"
    )

    # --------------------------------------------------------
    # Infer regime from observed behaviour
    # --------------------------------------------------------

    if (
        mean_shift > 2
        and change_probability >= 0.50
    ):

        regime = "Peak Shock"

    elif (
        change_probability
        >= CHANGE_THRESHOLD
    ):

        regime = "Structural Shift"

    elif (
        volatility_ratio
        > 1.50
    ):

        regime = "High Variability"

    else:

        regime = "Normal Demand"

    print(
        f"    CURRENT REGIME   : "
        f"{regime}"
    )

    return {

        "regime":
            regime,

        "mean_shift":
            mean_shift,

        "volatility_ratio":
            volatility_ratio
    }

# 6. DYNAMIC MODEL ROUTER
# ============================================================

def model_router(data):

    print("\n[5] DYNAMIC MODEL ROUTING")

    features = [

        "lag_1",

        "lag_7",

        "lag_14",

        "rolling_mean_7",

        "rolling_std_7",

        "day_of_week",

        "month"
    ]

    X = data[features]

    y = data[TARGET]

    validation_size = 20

    X_train = (
        X.iloc[:-validation_size]
    )

    y_train = (
        y.iloc[:-validation_size]
    )

    X_test = (
        X.iloc[-validation_size:]
    )

    y_test = (
        y.iloc[-validation_size:]
    )

    # ========================================================
    # MODEL 1 — LINEAR REGRESSION
    # ========================================================

    linear_model = (
        LinearRegression()
    )

    linear_model.fit(
        X_train,
        y_train
    )

    linear_predictions = (
        linear_model.predict(
            X_test
        )
    )

    linear_mae = (
        mean_absolute_error(
            y_test,
            linear_predictions
        )
    )

    print(
        f"    Linear Regression MAE : "
        f"{linear_mae:.4f}"
    )

    # ========================================================
    # MODEL 2 — ADAPTIVE BASELINE
    # ========================================================

    baseline_value = (
        y_train.iloc[-7:].mean()
    )

    baseline_predictions = (
        np.repeat(
            baseline_value,
            validation_size
        )
    )

    baseline_mae = (
        mean_absolute_error(
            y_test,
            baseline_predictions
        )
    )

    print(
        f"    Adaptive Baseline MAE  : "
        f"{baseline_mae:.4f}"
    )

    # ========================================================
    # ROUTING DECISION
    # ========================================================

    if (
        linear_mae
        <=
        baseline_mae
    ):

        selected_model = (
            "Linear Regression"
        )

        final_model = (
            LinearRegression()
        )

        final_model.fit(
            X,
            y
        )

        latest_X = (
            X.iloc[[-1]]
        )

        forecast = float(
            final_model.predict(
                latest_X
            )[0]
        )

    else:

        selected_model = (
            "Adaptive Baseline"
        )

        forecast = float(
            y.iloc[-7:].mean()
        )

    print(
        f"    SELECTED MODEL       : "
        f"{selected_model}"
    )

    print(
        f"    FORECAST             : "
        f"{forecast:.3f}"
    )

    return {

        "selected_model":
            selected_model,

        "forecast":
            forecast,

        "linear_mae":
            linear_mae,

        "baseline_mae":
            baseline_mae
    }

# 7. ADAPTIVE CONFORMAL PREDICTION
# ============================================================

def adaptive_conformal(
    data,
    forecast
):

    print(
        "\n[6] ADAPTIVE CONFORMAL PREDICTION"
    )

    values = (
        data[TARGET]
        .values
    )

    errors = []

    start = max(
        7,
        len(values) - 60
    )

    # --------------------------------------------------------
    # Historical one-step errors
    # --------------------------------------------------------

    for i in range(
        start,
        len(values)
    ):

        if i < 7:
            continue

        actual = (
            values[i]
        )

        previous_values = (
            values[i - 7:i]
        )

        prediction = (
            np.mean(
                previous_values
            )
        )

        error = abs(
            actual
            -
            prediction
        )

        errors.append(
            error
        )

    errors = np.asarray(
        errors
    )

    if len(errors) == 0:

        adaptive_radius = np.std(
            values
        )

    else:

        conformal_quantile = (
            np.quantile(
                errors,
                COVERAGE_LEVEL
            )
        )

        recent_volatility = (
            np.std(
                values[-14:]
            )
        )

        # Adapt interval width to recent behaviour
        adaptive_radius = (
            0.70
            *
            conformal_quantile
            +
            0.30
            *
            recent_volatility
        )

    lower = (
        forecast
        -
        adaptive_radius
    )

    upper = (
        forecast
        +
        adaptive_radius
    )

    print(
        f"    Target coverage    : "
        f"{COVERAGE_LEVEL * 100:.0f}%"
    )

    print(
        f"    Calibration errors : "
        f"{len(errors)}"
    )

    print(
        f"    Adaptive radius    : "
        f"{adaptive_radius:.3f}"
    )

    print(
        f"    Prediction interval: "
        f"{lower:.3f} → {upper:.3f}"
    )

    return {

        "lower":
            lower,

        "upper":
            upper,

        "radius":
            adaptive_radius,

        "coverage":
            COVERAGE_LEVEL
    }

# 8. CONFIDENCE SCORE
# ============================================================

def calculate_confidence(
    change_probability,
    regime,
    model_mae,
    interval_radius,
    forecast
):

    # Stability
    stability = (
        1
        -
        change_probability
    )

    # Regime reliability
    regime_scores = {

        "Normal Demand":
            1.00,

        "High Variability":
            0.70,

        "Peak Shock":
            0.50,

        "Structural Shift":
            0.55
    }

    regime_component = (
        regime_scores.get(
            regime,
            0.50
        )
    )

    # Forecast accuracy
    scale = max(
        abs(forecast),
        1.0
    )

    accuracy_component = (
        1
        /
        (
            1
            +
            model_mae / scale
        )
    )

    # Uncertainty
    uncertainty_component = (
        1
        /
        (
            1
            +
            interval_radius / scale
        )
    )

    score = (

        0.35
        *
        stability

        +

        0.25
        *
        regime_component

        +

        0.25
        *
        accuracy_component

        +

        0.15
        *
        uncertainty_component
    )

    confidence_score = float(
        np.clip(
            score * 100,
            0,
            100
        )
    )

    if confidence_score >= 75:

        label = "HIGH"

    elif confidence_score >= 50:

        label = "MODERATE"

    else:

        label = "LOW"

    return (
        confidence_score,
        label
    )

# 10. MAIN PIPELINE
# ============================================================

def main():

    # ========================================================
    # DATA
    # ========================================================

    df = load_data()

    if df is None:
        return

    # ========================================================
    # FEATURES
    # ========================================================

    data = create_features(
        df
    )

    if len(data) < 100:

        print(
            "\nERROR: Not enough observations."
        )

        return

    # ========================================================
    # MODULE 1
    # CHANGE DETECTION
    # ========================================================

    detection = (
        run_streaming_change_detection(
            data
        )
    )

    history = (
        detection["history"]
    )

    first_change = (
        detection["first_change"]
    )

    if history:

        latest = history[-1]

        change_probability = (
            latest["probability"]
        )

    else:

        change_probability = 0.0

    # ========================================================
    # MODULE 2
    # REGIME DETECTION
    # ========================================================

    regime_result = (
        detect_regime(
            data[TARGET].values,
            change_probability
        )
    )

    # ========================================================
    # MODULE 3
    # MODEL ROUTING
    # ========================================================

    model_result = (
        model_router(
            data
        )
    )

    # ========================================================
    # MODULE 4
    # ADAPTIVE CONFORMAL
    # ========================================================

    conformal_result = (
        adaptive_conformal(
            data,
            model_result["forecast"]
        )
    )

    # ========================================================
    # CONFIDENCE
    # ========================================================

    confidence_score, confidence_label = (
        calculate_confidence(

            change_probability,

            regime_result["regime"],

            min(
                model_result["linear_mae"],
                model_result["baseline_mae"]
            ),

            conformal_result["radius"],

            model_result["forecast"]
        )
    )

    # ========================================================
    # UNCERTAINTY LEVEL
    # ========================================================

    interval_width = (
        conformal_result["upper"]
        -
        conformal_result["lower"]
    )

    forecast_scale = max(
        abs(
            model_result["forecast"]
        ),
        1.0
    )

    relative_width = (
        interval_width
        /
        forecast_scale
    )

    if relative_width < 0.25:

        uncertainty = "LOW"

    elif relative_width < 0.50:

        uncertainty = "MODERATE"

    else:

        uncertainty = "HIGH"

    # ========================================================
    # RESULT OBJECT
    # ========================================================

    result = {

        "forecast":
            model_result["forecast"],

        "lower":
            conformal_result["lower"],

        "upper":
            conformal_result["upper"],

        "radius":
            conformal_result["radius"],

        "coverage":
            conformal_result["coverage"],

        "change_probability":
            change_probability,

        "regime":
            regime_result["regime"],

        "selected_model":
            model_result["selected_model"],

        "linear_mae":
            model_result["linear_mae"],

        "baseline_mae":
            model_result["baseline_mae"],

        "confidence_score":
            confidence_score,

        "confidence_label":
            confidence_label,

        "uncertainty":
            uncertainty,

        "detection_history":
            history,

        "first_change":
            first_change,

        "genuine_change":
            first_change is not None
    }

    # ========================================================
    # TERMINAL OUTPUT
    # ========================================================

    print("\n")
    print("=" * 75)
    print("                         FINAL AERIS OUTPUT")
    print("=" * 75)
